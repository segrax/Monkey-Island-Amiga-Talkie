cmake_minimum_required(VERSION 3.20)
project(MonkeyIslandAmigaTalkie VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(cmake/EmbedAssets.cmake)
add_library(builder-core STATIC builder/formats.cpp builder/builder.cpp builder/planar.cpp builder/glyphs.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.cpp")
target_include_directories(builder-core PUBLIC builder)
target_include_directories(builder-core SYSTEM PUBLIC third_party)
add_executable(amiga-talkie builder/main.cpp)
target_link_libraries(amiga-talkie PRIVATE builder-core)
if(WIN32)
  target_link_libraries(amiga-talkie PRIVATE ole32 comdlg32 shell32 uuid)
  set_target_properties(amiga-talkie PROPERTIES OUTPUT_NAME "Build Amiga Talkie")
  if(MINGW)
    target_link_options(amiga-talkie PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
else()
  set_target_properties(amiga-talkie PROPERTIES OUTPUT_NAME build-amiga-talkie)
endif()
if(MSVC)
  target_compile_options(amiga-talkie PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(builder-core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(amiga-talkie PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(builder-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME embedded_payload COMMAND amiga-talkie --verify-payload)
  add_test(NAME format_selftest COMMAND amiga-talkie --self-test)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  find_package(Threads REQUIRED)
  add_executable(builder-test-driver tests/builder_driver.cpp)
  target_link_libraries(builder-test-driver PRIVATE builder-core Threads::Threads)
  add_test(NAME builder_lifecycle COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/builder_lifecycle.py"
    "$<TARGET_FILE:builder-test-driver>" "$<TARGET_FILE:amiga-talkie>")
  set_tests_properties(builder_lifecycle PROPERTIES TIMEOUT 60)
endif()
install(TARGETS amiga-talkie RUNTIME DESTINATION .)
