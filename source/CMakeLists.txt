cmake_minimum_required(VERSION 3.20)
project(MonkeyIslandAmigaTalkie VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(cmake/EmbedAssets.cmake)
add_executable(amiga-talkie builder/main.cpp builder/formats.cpp builder/builder.cpp builder/planar.cpp builder/glyphs.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.cpp")
target_include_directories(amiga-talkie PRIVATE builder)
target_include_directories(amiga-talkie SYSTEM PRIVATE third_party)
if(WIN32)
  target_link_libraries(amiga-talkie PRIVATE ole32 comdlg32 shell32 uuid)
  set_target_properties(amiga-talkie PROPERTIES OUTPUT_NAME "Build Amiga Talkie")
  if(MINGW)
    target_link_options(amiga-talkie PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
else()
  set_target_properties(amiga-talkie PROPERTIES OUTPUT_NAME build-amiga-talkie)
endif()
if(MSVC)
  target_compile_options(amiga-talkie PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(amiga-talkie PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME embedded_payload COMMAND amiga-talkie --verify-payload)
  add_test(NAME format_selftest COMMAND amiga-talkie --self-test)
endif()
install(TARGETS amiga-talkie RUNTIME DESTINATION .)
